💬 متدهای مدیریت پیام‌ها و تاریخچه چت Messages API

NEW await bot.get_chat_history(guid, limit=50)
List[Message]

متد اختصاصی و هوشمند تاریخچه پیام‌ها: این متد با ترکیب اندپوینت‌های پیام و آپدیت و با بهره‌گیری از صفحه‌بندی خودکار (Automatic Pagination)، تاریخچه یک گفتگو (کاربر، گروه یا کانال) را تا سقف مقدار درخواستی دریافت کرده و به‌صورت لیستی از اشیای کامل Message برمی‌گرداند.

نام پارامتر نوع پیش‌فرض توضیحات
guid / object_guid str اجباری شناسه اختصاصی کاربر (u0...)، گروه (g0...) یا کانال (c0...)
limit int 50 تعداد پیام‌های درخواستی (با پشتیبانی از مقادیر بزرگ و دریافت دسته‌ای)
sort str "FromMax" مرتب‌سازی پیام‌ها بر اساس شناسه ("FromMax" یا "FromMin")
Python — get_chat_history Example 
from shadism import Client

bot = Client("0937xxxxxxx")

@bot.on_message
async def history_example(message):
    if message.text == "تاریخچه":
        # دریافت ۲۰ پیام اخیر با تمام امکانات شیء‌گرا
        history = await bot.get_chat_history(message.chat_guid, limit=20)
        
        for msg in history:
            print(f"[{msg.id}] {msg.author_guid}: {msg.text}")
            
            # متدهای مستقیم روی تک تک پیام‌ها
            if "سلام" in (msg.text or ""):
                await msg.reply("درود!")

🎯 سیستم قدرتمند فیلترها (Event Filters) Filters

MODULE shadism.filters — Declarative Filter Engine
Filter

سیستم فیلتر ماژولار با پشتیبانی از عملگرهای منطقی & (AND)، | (OR) و ~ (NOT) برای پردازش انتخابی پیام‌ها.

فیلتر عملکرد نمونه استفاده
filters.command تشخیص دستورات با پیشوندهای دلخواه filters.command("start", prefixes=["/", "!"])
filters.regex تطبیق با الگوی منظم (Regular Expressions) filters.regex(r"^دانلود\s+(.+)")
filters.private تنها پیام‌های چت‌های خصوصی (PV) filters.private
filters.group تنها پیام‌های ارسال‌شده در گروه‌ها filters.group
filters.channel تنها پیام‌های کانال‌ها filters.channel
filters.reply پیام‌هایی که در پاسخ به پیام دیگری هستند filters.reply
filters.create ساخت فیلتر شرطی دلخواه و تابع سفارشی filters.create(lambda m: "سلام" in m.text)
Python — Filter Composition 
from shadism import Client, filters

bot = Client("0937xxxxxxx")

# فقط دستور پینگ در گروه‌ها
@bot.on_message(filters.command("ping") & filters.group)
async def ping_in_group(message):
    await message.reply("پونگ در گروه! 🏓")

# ترکیب دستورات چندگانه در پی‌وی
@bot.on_message(filters.command(["help", "راهنما"]) & filters.private)
async def private_help(message):
    await message.reply("راهنمای چت خصوصی...")

import asyncio
from shadism import Client, filters

bot = Client("0937xxxxxxx")

@bot.on_message(filters.command("start"))
async def start_cmd(message):
    await message.reply(
        "🤖 به یوزربات شادیسم خوش آمدید!\n\n"
        "1. ارسال تصویر\n"
        "2. پاکسازی دوطرفه پیام‌های من\n"
        "3. تاریخچه گفتگو"
    )

@bot.on_message(filters.command("history") | filters.command("تاریخچه"))
async def chat_history_cmd(message):
    history = await bot.get_chat_history(message.chat_guid, limit=10)
    lines = [f"📜 آخرین {len(history)} پیام چت:"]
    for i, m in enumerate(history, 1):
        lines.append(f"{i}. {m.text[:30]} | فرستنده: {m.author_guid[:8]}...")
    await message.reply("\n".join(lines))

@bot.on_message(filters.command("clean"))
async def clean_cmd(message):
    # پیدا کردن پیام‌ها با get_chat_history و پاکسازی دوطرفه
    history = await bot.get_chat_history(message.chat_guid, limit=50)
    my_messages = [m.id for m in history if m.author_guid == bot.session.user_guid]
    
    if my_messages:
        await bot.delete_messages(message.chat_guid, my_messages, delete_type="Global")
        await message.reply(f"🗑️ تعداد {len(my_messages)} پیام دوطرفه پاکسازی شد.")

async def main():
    await bot.start()

if __name__ == "__main__":
    asyncio.run(main())
